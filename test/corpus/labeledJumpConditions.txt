const _: () = 'a: while break 'a {};
fn conditions() {
    'outer: loop {
        if break 'outer { if_body(); }
        while break 'outer { while_body(); }
        while continue 'outer { continue_body(); }
        match break 'outer { _ => {} }
    }
    after();
}

    let block = 'value: loop { break 'value { 1 }; };
    let character = loop { break 'x'; };
    let number = 'number: loop { break 'number 2; };
    after();
