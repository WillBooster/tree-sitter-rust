#![feature(explicit_tail_calls)]
#![allow(incomplete_features, dead_code)]
fn identity(value: u32) -> u32 { value }
fn recurse(value: u32) -> u32 {
    if value == 0 { 0 } else { become recurse(value - 1) }
}
fn through_pointer(value: u32) -> u32 {
    let target: fn(u32) -> u32 = identity;
    become target(value);
}
fn through_table(value: u32) -> u32 {
    let index = 0;
    let table: [fn(u32) -> u32; 1] = [identity];
    become table[index](value)
}
macro_rules! tail { ($call:expr) => { become $call }; }
fn through_macro(value: u32) -> u32 { tail!(identity(value)) }
fn r#become() {}
fn raw_name() { r#become(); }
