#![feature(const_trait_impl)]
#![allow(dead_code)]
pub const trait Value { fn value(&self) -> u8; }
const impl Value for u8 { fn value(&self) -> u8 { *self } }
struct Wrapper<T>(T);
const impl<T> Value for Wrapper<T> { fn value(&self) -> u8 { 2 } }
pub(crate) const unsafe trait Marker {}
const unsafe impl<T> Marker for Wrapper<T> {}
const VALUE: u8 = 1u8.value() + Wrapper(2u8).value();
const fn ordinary() -> u8 { VALUE }
pub const unsafe fn unsafe_function() -> u8 { VALUE }
fn main() { assert_eq!(ordinary(), 3); }
